    90	            cells = [cell.strip() for cell in re.split(r"(?<!\\)\|", line)[1:-1]]
    91	            assert len(cells) == 4, line
    92	            checks.append(dict(zip(("group", "item", "status", "detail", "proof"),
    93	                                   [group, *cells])))
    94	    return checks
