## Example (one test in the expected shape; copy the shape, not the content)

```python
def test_locale_does_not_change_bytes(tmp_path):
    """B3.3 — the same entries must give byte-identical INDEX.md under two different locales."""
    repo = make_repo(tmp_path)
    for name in ("D-b.md", "D-B.md", "D-é.md", "D-a b.md"):
        write_entry(repo, name, id=name[:-3], kind="decision", status="active")
    git(repo, "add", "-A")
    run_index(repo, env_extra={"LC_ALL": "C"})
    first = (repo / "project-memory/INDEX.md").read_bytes()
    run_index(repo, env_extra={"LC_ALL": "C.UTF-8"})
    assert (repo / "project-memory/INDEX.md").read_bytes() == first
```
